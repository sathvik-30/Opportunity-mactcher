# Authentication
# bcrypt is pinned <4.0 — passlib 1.7.4's backend-detection code breaks
# against bcrypt>=4.0 (it reads bcrypt.__about__.__version__, removed in
# 4.0), causing every hash/verify call to fail with a bogus "password
# cannot be longer than 72 bytes" error from passlib's own internal
# self-test. This is the standard fix until passlib releases a fix.
passlib[bcrypt]
bcrypt==3.2.2
python-jose[cryptography]
python-multipart
